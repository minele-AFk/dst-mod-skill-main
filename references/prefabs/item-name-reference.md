```lua

-- 代码名: hambat

```

## 查找工具函数

可以在mod中添加以下函数来快速查找物品名称：

```lua
local ITEM_NAMES = {}

local function LoadItemNames()
    local file_path = "mods/.trae/skills/dst-mod-skill-main/references/prefabs/物品名字.txt"
    local success, data = pcall(dofile, file_path)
    if success and type(data) == "table" then
        ITEM_NAMES = data
    end
end

local function GetItemChineseName(code_name)
    return ITEM_NAMES[code_name] or code_name
end

local function FindItemCode(chinese_name)
    for code, name in pairs(ITEM_NAMES) do
        if name == chinese_name then
            return code
        end
    end
    return nil
end

LoadItemNames()
```

使用示例：

```lua
-- 获取中文名称
print(GetItemChineseName("hambat"))  -- 输出: 火腿棒

-- 查找代码名
print(FindItemCode("火腿棒"))  -- 输出: hambat
```
